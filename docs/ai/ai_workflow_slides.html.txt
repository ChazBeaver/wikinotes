(function () {
  var deck = document.getElementById('deck');
  var slides = Array.prototype.slice.call(document.querySelectorAll('.slide'));
  var notes = document.getElementById('notes');
  var counter = document.getElementById('counter');
  var prog = document.getElementById('prog');
  var grid = document.getElementById('grid');
  var n = slides.length;
  var cur = 0;

  slides.forEach(function (s, i) {
    var num = s.querySelector('.num');
    if (num) num.textContent = (i + 1) + ' / ' + n;
    var b = document.createElement('button');
    b.setAttribute('role', 'listitem');
    var bn = document.createElement('span'); bn.className = 'n'; bn.textContent = String(i + 1);
    var bt = document.createElement('span'); bt.className = 't';
    bt.textContent = s.getAttribute('data-title') || ('Slide ' + (i + 1));
    b.appendChild(bn); b.appendChild(bt);
    b.addEventListener('click', function () { deck.classList.remove('overview'); go(i); });
    grid.appendChild(b);
  });

  function go(i) {
    cur = Math.max(0, Math.min(n - 1, i));
    slides.forEach(function (s, k) { s.classList.toggle('is-active', k === cur); });
    counter.textContent = (cur + 1) + ' / ' + n;
    prog.max = n;
    prog.value = cur + 1;
    var a = slides[cur].querySelector('aside[data-notes]');
    notes.replaceChildren();
    if (a) Array.prototype.forEach.call(a.childNodes, function (node) { notes.appendChild(node.cloneNode(true)); });
    Array.prototype.forEach.call(grid.children, function (b, k) { b.classList.toggle('is-current', k === cur); });
    try { history.replaceState(null, '', '#' + (cur + 1)); } catch (e) {}
    try { localStorage.setItem('howai.slide', String(cur)); } catch (e) {}
  }

  var start = 0;
  var h = parseInt(location.hash.slice(1), 10);
  if (h >= 1 && h <= n) start = h - 1;
  else { try { var s = parseInt(localStorage.getItem('howai.slide'), 10); if (s >= 0 && s < n) start = s; } catch (e) {} }
  go(start);

  document.getElementById('btnPrev').addEventListener('click', function () { go(cur - 1); });
  document.getElementById('btnNext').addEventListener('click', function () { go(cur + 1); });

  document.addEventListener('keydown', function (e) {
    if (e.target && /^(INPUT|TEXTAREA)$/.test(e.target.tagName)) return;
    switch (e.key) {
      case 'ArrowRight': case ' ': case 'PageDown': e.preventDefault(); go(cur + 1); break;
      case 'ArrowLeft': case 'PageUp': e.preventDefault(); go(cur - 1); break;
      case 'Home': go(0); break;
      case 'End': go(n - 1); break;
      case 's': case 'S': deck.classList.toggle('show-notes'); break;
      case 'o': case 'O': deck.classList.toggle('overview'); break;
      case 'p': case 'P': window.print(); break;
      case 'Escape': deck.classList.remove('overview'); break;
    }
  });
  window.addEventListener('hashchange', function () {
    var k = parseInt(location.hash.slice(1), 10);
    if (k >= 1 && k <= n && k - 1 !== cur) go(k - 1);
  });
})();
